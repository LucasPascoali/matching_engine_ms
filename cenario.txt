# ==================================================================
# Cenario de teste ponta-a-ponta para CLI.py
# Uso: python -m engine < cenario.txt
#
# Cobre: help, book vazio, FIFO dentro do mesmo nivel, crossing de
# limit (parcial e atravessando multiplos niveis), market como IOC
# (fill total e fill parcial com descarte do resto), cancel
# (sucesso e id inexistente), modify (reduz qty mantem prioridade,
# aumenta qty perde prioridade, muda preco perde prioridade, muda
# preco causando crossing, guard contra price em pegged, modify de
# id inexistente/ja cancelado), pegged orders (pendente sem
# referencia, precificacao ao surgir referencia, estabilidade
# quando ainda ha ordem real no nivel, virar pendente quando so
# sobra pegged no nivel - cenario do bug original -, multiplas
# pegged no mesmo nivel, peg to offer) e validacoes de erro do CLI.
# ==================================================================

help

# --- book vazio ---
print book

# --- FIFO dentro do mesmo nivel, sem crossing ---
limit buy 10.0 100
limit buy 10.0 50
limit sell 11.0 80
print book

# --- limit que cruza, fill parcial da primeira ordem da fila (FIFO) ---
limit sell 10.0 30
print book

# --- limit que atravessa multiplas ordens no mesmo nivel e sobra, vira resting ---
limit sell 9.5 150
print book

# --- ordem que nao cruza, so fica resting ---
limit buy 8.0 40
print book

# --- market (IOC) que cruza multiplos niveis e fica totalmente preenchida ---
market buy 50
print book

# --- market (IOC) com liquidez insuficiente: preenche o que der e descarta o resto ---
market sell 100
print book

# --- cancel: sucesso ---
cancel order ID_3
print book

# --- cancel: id inexistente ---
cancel order ID_999

# --- modify: reduzir qty mantem prioridade ---
limit buy 10.0 100
limit buy 10.0 100
print book
modify ID_9 qty=40
print book

# --- modify: aumentar qty perde prioridade (cancel/replace) ---
modify ID_9 qty=200
print book

# --- modify: mudar preco perde prioridade ---
modify ID_10 price=9.0
print book
modify ID_10 price=10.0
print book

# --- modify: id inexistente ---
modify ID_999 qty=10

# --- modify que causa crossing na hora ---
limit sell 10.0 50
print book

limit sell 12.0 30
print book
modify ID_12 price=10.0
print book

# --- pegged: guard contra price manual ---
peg bid buy 50
print book
modify ID_13 price=9.0
print book

# --- pegged: reprice estavel enquanto ainda ha ordem real no nivel ---
limit sell 9.5 20
print book

# --- pegged: remove ordens reais uma a uma; pegged deve ficar pendente
#     quando so sobra ela (ou outras pegged) no nivel - bug original ---
cancel order ID_9
print book
cancel order ID_10
print book

# --- pegged: reprecifica quando surge nova referencia real ---
limit buy 7.0 200
print book

# --- pegged: multiplas pegged no mesmo nivel ---
peg bid buy 30
print book

# --- pegged: remover a unica ordem real deixa as pegged pendentes de novo ---
cancel order ID_15
print book

# --- pegged: nova liquidez reprecifica ambas; peg to offer tambem ---
limit buy 6.5 60
limit sell 8.0 40
print book
peg offer sell 25
print book

# --- validacoes de erro no parsing de comandos ---
limit buy 10
market buy 10 5.0
limit foo 5.0 10
weird buy 10 5.0
peg buy 10
cancel order
modify

# --- modify em ordem ja cancelada ---
modify ID_9 price=10

# --- estado final ---
print book

quit
